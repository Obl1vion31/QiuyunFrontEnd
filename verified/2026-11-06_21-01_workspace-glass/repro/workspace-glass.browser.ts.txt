import assert from 'node:assert/strict';
import {mkdir,writeFile} from 'node:fs/promises';
import {startPostgresFixture} from '../tests/postgres.fixture';
import {admin,employee} from '../tests/auth.fixture';
import {settleBusinessUI} from '../tests/business-confirm.helper';
import {seedWorkspaceGlass,CLOCK} from './workspace-glass.fixture';
const phase=process.env.GLASS_PHASE??'before',folder='.tools/test-artifacts/workspace-glass/'+phase;
await mkdir(folder+'/screenshots',{recursive:true});await writeFile(folder+'/clock.txt',CLOCK);
const f=await startPostgresFixture({clockFile:process.cwd()+'/'+folder+'/clock.txt'});
let browser:any,p:any;const errors:string[]=[],geometry:any[]=[],content:any={};
async function login(actor:any){const r=await fetch(f.baseURL+'/api/auth/sign-in/username',{method:'POST',headers:{Origin:f.baseURL,'Content-Type':'application/json'},body:JSON.stringify(actor)});assert.equal(r.status,200);return r.headers.getSetCookie().map(v=>{const[name,...rest]=v.split(';')[0].split('=');return{name,value:rest.join('='),url:f.baseURL};});}
try{
 const sample=await seedWorkspaceGlass(f.db);
 await writeFile('.tools/workspace-preview-'+phase+'.json',JSON.stringify({pid:process.pid,baseURL:f.baseURL,phase,routes:sample.routes,digest:sample.digest,clock:CLOCK},null,2));
 const {chromium}=await import(process.env.PLAYWRIGHT_MODULE!);browser=await chromium.launch({headless:true});
 const cookies=await login(admin),staff=await login(employee);
 for(const width of [1440,1920,1024]){
  const c=await browser.newContext({viewport:{width,height:1000},locale:'zh-CN'});await c.addCookies(cookies);p=await c.newPage();p.on('pageerror',(e:Error)=>errors.push(e.message));await p.clock.setFixedTime(new Date(CLOCK));
  for(const [name,path]of Object.entries(sample.routes)){
   if(name==='login')continue;
   await p.goto(f.baseURL+path);await p.waitForLoadState('networkidle');await settleBusinessUI(p);
   if(name==='home'){await p.emulateMedia({reducedMotion:'reduce'});await p.reload();await settleBusinessUI(p);}
   if(['monthly','operations','weekly'].includes(name)){
    for(const chart of await p.locator('[data-report-chart],[data-monthly-chart]').all()){await chart.scrollIntoViewIfNeeded();await settleBusinessUI(p);}await p.evaluate(()=>scrollTo(0,0));
   }
   await p.mouse.move(0,0);await p.screenshot({path:folder+'/screenshots/'+name+'-'+width+'.png'});
   if(width===1440)await p.screenshot({path:folder+'/screenshots/'+name+'-'+width+'-full.png',fullPage:true});
   geometry.push(await p.evaluate((name:string)=>({name,width:innerWidth,scroll:document.documentElement.scrollWidth}),name));
   if(width===1440)content[name]=await p.locator('main').innerText();
   if(name==='home')await p.emulateMedia({reducedMotion:'no-preference'});
   if(name==='monthly'){
    const panel=p.locator('[data-monthly-meeting-overview]');await panel.scrollIntoViewIfNeeded();await settleBusinessUI(p);await p.screenshot({path:folder+'/screenshots/monthly-execution-'+width+'.png'});
   }
   if(name==='promotion-history'){
    await p.locator('.promotion-data-disclosure>summary').first().click();await settleBusinessUI(p);const row=p.locator('tr[data-inspector-key]:visible').first();
    if(await row.count()){await row.click();await settleBusinessUI(p);await p.screenshot({path:folder+'/screenshots/promotion-inspector-'+width+'.png'});}
   }
   if(name==='nonpromotion-history'){
    const group=p.locator('[data-category-group]').first();if(await group.count()){await group.locator('summary').first().click();await settleBusinessUI(p);await p.screenshot({path:folder+'/screenshots/nonpromotion-expanded-'+width+'.png'});}
   }
  }
  await c.close();
  const anon=await browser.newContext({viewport:{width,height:1000},locale:'zh-CN'});p=await anon.newPage();await p.goto(f.baseURL+'/login');await settleBusinessUI(p);await p.screenshot({path:folder+'/screenshots/login-'+width+'.png'});await anon.close();
 }
 const c=await browser.newContext({viewport:{width:1440,height:1000},locale:'zh-CN',reducedMotion:'reduce'});await c.addCookies(staff);p=await c.newPage();await p.goto(f.baseURL+sample.routes.nonpromotion);await settleBusinessUI(p);await p.screenshot({path:folder+'/screenshots/employee-1440.png'});await c.close();
 await writeFile(folder+'/result.json',JSON.stringify({phase,clock:CLOCK,digest:sample.digest,errors,geometry,content},null,2));
 console.log(JSON.stringify({phase,baseURL:f.baseURL,errors,count:geometry.length,routes:sample.routes},null,2));
 await browser.close();browser=null;
 if(process.env.GLASS_KEEP==='1')await new Promise<void>(resolve=>{process.once('SIGTERM',resolve);process.once('SIGINT',resolve);});
}catch(e){if(p&&!p.isClosed())await p.screenshot({path:folder+'/debug.png'});console.error(f.logs().slice(-2500));throw e;}finally{await browser?.close();await f.stop();}
