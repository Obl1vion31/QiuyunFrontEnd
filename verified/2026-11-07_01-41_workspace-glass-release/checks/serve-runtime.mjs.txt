import {createServer} from 'node:http';import {readFile,stat} from 'node:fs/promises';import {resolve,extname} from 'node:path';import handler from '../.vercel/output/functions/_render.func/dist/server/entry.mjs';
const root=resolve('.vercel/output/static'),types={'.js':'application/javascript','.css':'text/css','.html':'text/html','.woff2':'font/woff2','.svg':'image/svg+xml','.png':'image/png'};
createServer(async(req,res)=>{try{const url=new URL(req.url,'http://127.0.0.1'),path=resolve(root,'.'+decodeURIComponent(url.pathname==='/'?'/index.html':url.pathname));if(path.startsWith(root+'/'))try{if((await stat(path)).isFile()){res.writeHead(200,{'Content-Type':types[extname(path)]??'application/octet-stream'});res.end(await readFile(path));return;}}catch{}req.headers['x-forwarded-proto']='http';await handler(req,res);}catch(error){console.error(error);res.writeHead(500);res.end('Runtime verification failed');}}).listen(Number(process.env.PORT),'127.0.0.1');
